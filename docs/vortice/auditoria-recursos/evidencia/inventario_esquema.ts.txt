import fs from "node:fs"; import os from "node:os"; import path from "node:path";
const dir = fs.mkdtempSync(path.join(os.tmpdir(), "inv-")); process.env.TEST_DATABASE_URL = `file:${path.join(dir, "a.db")}`;
const { sql } = await import("drizzle-orm");
const { generateSQLiteDrizzleJson, generateSQLiteMigration } = await import("drizzle-kit/api");
const S = await import("../shared/schema"); const { db } = await import("../server/db");
for (const st of await generateSQLiteMigration(await generateSQLiteDrizzleJson({}), await generateSQLiteDrizzleJson(S as any))) await db.run(sql.raw(st));
const c: any = (db as any).$client; const { migrar } = await import("../server/migrar"); await migrar(c, "migrations/vortice");
const rel = ["levels","subjects","level_subjects","learning_objectives","weekly_resources","module_evaluations","evaluation_progress","student_progress","reflection_responses","review_items","learning_events","portfolio_evidences","weekly_completion","rubrics","tutoring_cases","brujula_weeks","textbook_configs","enrollments","curriculum_objectives","knowledge_domains","knowledge_domain_versions","knowledge_items","knowledge_item_versions","knowledge_prerequisites","knowledge_item_oas","knowledge_item_modules","knowledge_item_resources","question_families","question_versions","question_voids"];
const out: string[] = [];
for (const t of rel) {
  const cols = (await c.execute(`PRAGMA table_info(${t})`)).rows;
  const fks = (await c.execute(`PRAGMA foreign_key_list(${t})`)).rows;
  const idx = (await c.execute(`PRAGMA index_list(${t})`)).rows.filter((r: any) => !String(r.name).startsWith("sqlite_autoindex"));
  const un = (await c.execute(`PRAGMA index_list(${t})`)).rows.filter((r: any) => Number(r.unique) === 1);
  const trg = (await c.execute(`SELECT name FROM sqlite_master WHERE type='trigger' AND tbl_name='${t}' ORDER BY name`)).rows.map((r: any) => r.name);
  const refIn = [] as string[];
  for (const o of rel) for (const f of (await c.execute(`PRAGMA foreign_key_list(${o})`)).rows) if (f.table === t) refIn.push(`${o}.${f.from}`);
  out.push(`## ${t}\ncols: ${cols.map((r: any) => `${r.name} ${r.type}${r.notnull ? " NN" : ""}${r.pk ? " PK" : ""}${r.dflt_value != null ? " =" + r.dflt_value : ""}`).join("; ")}\nfk: ${fks.map((f: any) => `${f.from}->${f.table}.${f.to}(${f.on_delete})`).join("; ") || "-"}\nunique: ${un.length}; indices: ${idx.map((r: any) => r.name).join(",") || "-"}\ntriggers(${trg.length}): ${trg.join(",") || "-"}\nentrantes: ${refIn.join(", ") || "-"}`);
}
fs.writeFileSync(process.argv[2], out.join("\n\n"));
console.log("tablas", rel.length); process.exit(0);
